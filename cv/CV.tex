\section{Research Experience}

\begin{cventries}
  \roleentry{Lower Semicontinuity under Variable-Coefficient Differential Constraints}
            {Spring 2026 -- present}
            {Carnegie Mellon University.\factsep Research advisor:
             \href{https://irenefonseca.weebly.com}{Irene Fonseca}.}
\begin{cvbullets}
  \item Characterizing weak lower semicontinuity under linear differential
    constraints with variable coefficients via frozen $\A$-quasiconvexity,
    beyond setting of smooth coefficients with uniform constant rank.
  \item Proved sufficiency for Lipschitz coefficients under a pointwise
    constant-rank condition, allowing the rank to vary with $x$, via blow-up
    and projections onto the kernels of the frozen operators.
  \item Proved necessity at every point near which, for each lattice direction
    $\xi$, the rank of the symbol $\mathbb{A}(x,\xi)$ is locally constant in $x$,
    adapting the Davoli--Fonseca moving-kernel construction in place of
    pseudodifferential calculus.
\end{cvbullets}

  \roleentry{Tug-of-War Games and the Regularized $p$-Laplacian}
            {Fall 2025 -- present}
            {M.S.\ thesis research, University of Pittsburgh.\factsep Advisor:
             \href{https://sites.pitt.edu/~manfredi/}{Juan J.\ Manfredi}.}
  \begin{cvbullets}
    \item Observed that the regularized $p$-Laplacian of $u$ is the ordinary
          $p$-Laplacian of the affine lift of $u(x)$ in one more dimension, and
          computed the projected noise kernel and tilted extremal terms.
    \item Proved existence, uniqueness, and Borel measurability for the
          projected dynamic programming principle with boundary-strip data,
          via monotone iteration and an adapted comparison principle.
    \item Showed that the solution is the value of an explicit projected
          tug-of-war game on an augmented state space, using super- and
          submartingale arguments and measurable almost-optimal strategies.
    \item Proved convergence to the viscosity solution as the step size
          $\varepsilon\to 0$ for $p\ge 2$, and a pointwise asymptotic mean value
          characterization of $C^2$ solutions for every $p\in(1,\infty)$, which a
          counterexample shows fails at the formal endpoint $p=\infty$.
  \end{cvbullets}
\end{cventries}
